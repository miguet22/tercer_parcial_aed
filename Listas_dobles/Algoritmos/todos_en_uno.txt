			p := ult // recorrido de derecha a izq (del ultimo elemento al primero)
			mientras p <> nill hacer

			fin mientras
		fin si

		si prim = nill o ult = nill entonces

			esc ("ingrese un numero a buscar")
			leer (numusu)

			mientras p <> nill y *p.num <> numusu hacer 
				p := *p.prox
			fin mientras

			si p = nill entonces 
				esc ("error, no se encontro el numero a buscar")
			sino
				esc ("numero encontrado")
				// p queda parado en el nodo que posee el elemento a buscar
			fin si

			// busqueda lineal, que encuentra repetidos
			esc ("ingrese un numero a buscar")
			leer (numusu)
			cont := 0
			p := prim
			mientras p <> nill hacer
				si *p.num = numusu entonces 
					cont := cont + 1
				fin si 
				p := *p.prox
			fin mientras

			esc ("el numero a buscar se encontro:", cont "veces")

				// el numero se encontro hay que eliminarlo

					ult := nill
				sino

						// elimino el primero
						prim := *p.prox
						*(*p.prox).ant := nill
					sino

							// elimino el ultimo
							ult := *p.ant
							*(*p.ant).prox := nill
						sino
							// eliminar uno del medio

							*(*p.ant).prox := *p.prox
							*(*p.prox).ant := *p.ant
						fin si
					fin si
				fin si
				disponer (p)
			fin si
