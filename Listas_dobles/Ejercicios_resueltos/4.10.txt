Se dispone de una lista circular con un conjunto de números naturales. Este conjunto está dividido en 6 grupos. En cada grupo el primer elemento indica la cantidad de números subsiguientes a él que integran dicho grupo. Se desea generar una nueva lista doblemente encadenada en la cual cada nodo contendrá el promedio de cada grupo. Dicha lista debe quedar ordenada en forma ascendente y, al final se debe informar cuál fue el máximo y el mínimo promedio de todo el conjunto de números.


Accion 410 (prim : puntero a nodo) es 
	ambiente

	nodo = registro
		num : entero
		prox : puntero a nodo
	fin registro
	p : puntero a nodo // lista circ

	nodosal = registro
		prom : real
		prox,ant : puntero a nodosal
	fin registro
	prims,ults,ps,qs : puntero a nodosal
	sumatoria, cant_num : entero
	procedimiento carga_ordenada_listdoble () es 
	
		si prims = nill entonces 
			prims := qs
			ults := qs
			*qs.prox := nill
			*qs.ant := nill
		sino
			ps := prims

			mientras ps <> nill y *ps.prom < *qs.prom hacer 
				ps := *ps.prox
			fin mientras

			si ps = prims entonces 
				// camino 1 : al inicio
				*qs.prox := prims
				*prims.ant := qs
				*qs.ant := nill
				prims := qs

			sino
				si ps = nill entonces 
					// camino 2 : al final
					*qs.ant := ults
					*ults.prox := qs
					*qs.prox := nill
					ults := qs


				sino
					// camino 3 : en el medio
					*qs.prox := ps
					*qs.ant := *ps.ant
					*(*ps.ant).prox := qs
					*ps.ant := qs

				fin si
			fin si
		fin si

	fin procedimiento


	proceso
		si prim = nill entonces 
			esc ("error, lista de entrada vacia")
		sino
			p := prim
			prims := nill
			ults := nill
			mientras *p.prox <> prim hacer 

				// estan parados en el primer nodo del grupo (verde)

				sumatoria_num := 0
				cant_num := *p.num

				p := *p.prox
				para i := 1 a cant_num hacer 
					sumatoria_num := sumatoria_num + *p.num
					si *p.prox <> prim entonces // para no entrar en bucle infinito
						p := *p.prox
					fin si 
				fin para

				// estoy en el primer nodo de otro grupo

				nuevo (qs)
				*qs.prom := sumatoria_num / cant_num
				carga_ordenada_listdoble ()
			fin mientras

			// OPCION MAS CONFIABLE SEGUN LA IA 

			para g := 1 a 6 hacer
			    cant_num := *p.num
			    sumatoria_num := 0
			    para i := 1 a cant_num hacer
			        p := *p.prox
			        sumatoria_num := sumatoria_num + *p.num
			    fin para
			    // al salir del para interno, p apunta al último número del grupo actual.
			    // para posicionarnos en la cabecera del grupo siguiente:
			    p := *p.prox

			    nuevo (qs)
				*qs.prom := sumatoria_num / cant_num
				carga_ordenada_listdoble ()
			
			fin para

			
			esc ("MAXIMO PROMEDIO:", *ults.prom)
			esc ("MINIMO PROMEDIO:", *prims.prom)
		fin si
	fin proceso
fin accion
