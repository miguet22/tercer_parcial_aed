Accion listasimple_carga_apilada () es  
	ambiente

	nodo = registro
		// datos
		edad : entero
		
		// punteros internos
		prox: puntero a nodo
	fin registro
	// punteros externos
	prim2, p2, q2, ant2: puntero a nodo

	proceso

		esc ("desea cargar numeros")
		leer (rta)
		mientras rta = "si" hacer 
			nuevo (q2)
			esc ("INGRESE edad")
			leer (*q2.edad)

			// de aca
			
		
			*q2.prox := prim2
			prim2 := q2
			
			// hasta aca es lo que hay que saber, lo demas es de "adorno"
			esc ("desea cargar numeros")
			leer (rta)
		fin mientras
	fin proceso
fin accion

Accion listasimple_carga_encolada () es  
	ambiente

	nodo = registro
		// datos
		nya : alfanumerico
		
		// punteros internos
		prox: puntero a nodo
	fin registro
	// punteros externos
	prim2, p2, q2, ant2: puntero a nodo

	proceso

		esc ("desea cargar numeros")
		leer (rta)
		mientras rta = "si" hacer 
			nuevo (q2)
			esc ("INGRESE edad")
			leer (*q2.edad)

			// de aca
			*q2.prox := nill
			si prim2 = nill entonces 
				// lista vacia
				prim2 := q2
				
			sino
				*ant2.prox := q2
				
			fin si
			ant2 := q2

			// hasta aca es lo que hay que saber, lo demas es de "adorno"
			esc ("desea cargar numeros")
			leer (rta)
		fin mientras
	fin proceso
fin accion
